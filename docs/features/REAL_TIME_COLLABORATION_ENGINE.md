# 🔥 PartyHaus Real-Time Collaboration Engine

**Version:** 1.0  
**Date:** October 25, 2025  
**Feature:** Live Collaborative Planning System  
**Tagline:** "Plan together, decide together, celebrate together"

---

## 🎯 The Vision

**Problem:** Traditional event planning tools are just task lists with comments. They don't capture the **energy** of planning with friends - the debates, the excitement, the quick decisions, the creative sparks.

**PartyHaus Solution:** Transform event planning into a **live, interactive experience** where teams can:
- Vote on decisions in real-time
- React instantly to ideas
- Brainstorm visually together
- Debate options with pros/cons
- Make group decisions democratically
- See everyone's opinions as they form

## 📡 Technical Implementation

### **Real-Time Architecture:**

```typescript
// Supabase Realtime Channels
const eventChannel = supabase.channel(`event-${eventId}`);

// Subscribe to all collaboration events
eventChannel
  .on('postgres_changes', {
    event: 'INSERT',
    schema: 'public',
    table: 'polls'
  }, (payload) => {
    // New poll created
    showNotification('New poll created!');
    updatePollsList(payload.new);
  })
  .on('postgres_changes', {
    event: 'INSERT',
    schema: 'public',
    table: 'votes'
  }, (payload) => {
    // New vote cast
    updatePollResults(payload.new);
    animateVoteBar();
  })
  .on('postgres_changes', {
    event: 'INSERT',
    schema: 'public',
    table: 'reactions'
  }, (payload) => {
    // New reaction
    animateReaction(payload.new.emoji);
    updateReactionCount();
  })
  .on('broadcast', { event: 'cursor' }, (payload) => {
    // Live cursor positions
    updateCursor(payload.userId, payload.position);
  })
  .on('broadcast', { event: 'typing' }, (payload) => {
    // Typing indicators
    showTypingIndicator(payload.userId);
  })
  .on('presence', { event: 'sync' }, () => {
    // Who's online
    updatePresenceList();
  })
  .subscribe();
```

### **Database Schema Extensions:**

```sql
-- Polls table
CREATE TABLE polls (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  event_id UUID REFERENCES events(id),
  node_id UUID REFERENCES mind_map_nodes(id),
  created_by UUID REFERENCES users(id),
  
  question TEXT NOT NULL,
  poll_type VARCHAR(50) DEFAULT 'single-choice', -- 'single-choice', 'multiple-choice', 'ranking'
  
  options JSONB NOT NULL, -- [{ id, text, description }]
  
  ends_at TIMESTAMP,
  auto_close_on_consensus BOOLEAN DEFAULT TRUE,
  consensus_threshold INTEGER DEFAULT 70, -- percentage
  
  status VARCHAR(50) DEFAULT 'active', -- 'active', 'closed', 'consensus-reached'
  
  created_at TIMESTAMP DEFAULT NOW()
);

-- Votes table
CREATE TABLE votes (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  poll_id UUID REFERENCES polls(id) ON DELETE CASCADE,
  user_id UUID REFERENCES users(id),
  
  option_ids JSONB NOT NULL, -- Array of selected option IDs
  
  voted_at TIMESTAMP DEFAULT NOW(),
  
  UNIQUE(poll_id, user_id) -- One vote per poll per user
);

-- Reactions table
CREATE TABLE reactions (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  target_type VARCHAR(50) NOT NULL, -- 'node', 'comment', 'idea', 'vendor'
  target_id UUID NOT NULL,
  event_id UUID REFERENCES events(id),
  user_id UUID REFERENCES users(id),
  
  reaction_type VARCHAR(50) NOT NULL, -- emoji types
  
  created_at TIMESTAMP DEFAULT NOW(),
  
  UNIQUE(target_type, target_id, user_id, reaction_type)
);

-- Debates table
CREATE TABLE debates (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  event_id UUID REFERENCES events(id),
  node_id UUID REFERENCES mind_map_nodes(id),
  created_by UUID REFERENCES users(id),
  
  title TEXT NOT NULL,
  description TEXT,
  
  status VARCHAR(50) DEFAULT 'active', -- 'active', 'closed', 'decided'
  
  decision VARCHAR(50), -- 'pro', 'con', null
  decided_at TIMESTAMP,
  
  created_at TIMESTAMP DEFAULT NOW()
);

-- Debate points table
CREATE TABLE debate_points (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  debate_id UUID REFERENCES debates(id) ON DELETE CASCADE,
  user_id UUID REFERENCES users(id),
  
  point_type VARCHAR(10) NOT NULL, -- 'pro', 'con'
  content TEXT NOT NULL,
  
  upvotes INTEGER DEFAULT 0,
  downvotes INTEGER DEFAULT 0,
  
  created_at TIMESTAMP DEFAULT NOW()
);

-- Brainstorm sessions table
CREATE TABLE brainstorm_sessions (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  event_id UUID REFERENCES events(id),
  created_by UUID REFERENCES users(id),
  
  title TEXT NOT NULL,
  category VARCHAR(100),
  
  duration_minutes INTEGER DEFAULT 15,
  status VARCHAR(50) DEFAULT 'active', -- 'active', 'ended'
  
  started_at TIMESTAMP DEFAULT NOW(),
  ended_at TIMESTAMP
);

-- Ideas table (from brainstorming)
CREATE TABLE ideas (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  session_id UUID REFERENCES brainstorm_sessions(id) ON DELETE CASCADE,
  user_id UUID REFERENCES users(id),
  
  content TEXT NOT NULL,
  estimated_cost DECIMAL(10, 2),
  
  position_x INTEGER, -- For visual positioning
  position_y INTEGER,
  
  reaction_count INTEGER DEFAULT 0,
  
  converted_to_task BOOLEAN DEFAULT FALSE,
  task_id UUID REFERENCES mind_map_nodes(id),
  
  created_at TIMESTAMP DEFAULT NOW()
);

-- Collaboration points table
CREATE TABLE collaboration_points (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  event_id UUID REFERENCES events(id),
  user_id UUID REFERENCES users(id),
  
  action_type VARCHAR(50) NOT NULL, -- 'task_complete', 'vote_cast', 'idea_added', etc.
  points_earned INTEGER NOT NULL,
  
  reference_type VARCHAR(50), -- 'task', 'poll', 'idea', etc.
  reference_id UUID,
  
  earned_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_points_user_event ON collaboration_points(user_id, event_id);
CREATE INDEX idx_polls_event ON polls(event_id, status);
CREATE INDEX idx_votes_poll ON votes(poll_id);
CREATE INDEX idx_reactions_target ON reactions(target_type, target_id);
```
